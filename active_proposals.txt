```typescript
async function detectConflicts(directive: Proposal): Promise<ConflictReport> {
  // Search existing proposals for keyword overlap
  const keywords = extractKeywords(directive.title + ' ' + directive.summary);
  const candidates = await db.query(
    'SELECT id, title, summary FROM roadmap.proposals WHERE type != $1 AND id != $2',
    ['directive', directive.id]
  );
  
  const conflicts = [];
  for (const proposal of candidates) {
    const similarity = computeCosineSimilarity(keywords, extractKeywords(proposal.title + ' ' + proposal.summary));
    if (similarity > 0.6) {
      conflicts.push({
        proposal_id: proposal.id,
        similarity_score: similarity,
        requires_review: true
      });
    }
  }
  return { directive_id: directive.id, conflicts, timestamp: new Date() };
}
```

- REVIEW: D1-D2 gates evaluate feasibility, conflicts reviewed
- DEVELOP: Implementation phase, same as features
- MERGE: Branch integration, same quality gates
- COMPLETE: Stable state

### Priority Boost Implementation
In /data/code/AgentHive/src/dispatch/agent_dispatch.ts:

```typescript
async function selectProposalForAgent(): Proposal {
  const proposals = await db.query(`
    SELECT p.*, 
      CASE WHEN p.type = 'directive' THEN p.priority * 1.5 ELSE p.priority END AS dispatch_priority
    FROM roadmap.proposals p
    WHERE p.status = 'DEVELOP'
    ORDER BY dispatch_priority DESC, p.created_at ASC
    LIMIT 1
  `);
  return proposals[0] || null;
}
```

### Conflict Detection Trigger
In /data/code/AgentHive/src/mcp/prop_create.ts:

```typescript
export async function propCreate(args): Proposal {
  const proposal = await db.create('roadmap.proposals', args);
  
  if (proposal.type === 'directive') {
    const conflicts = await detectConflicts(proposal);
    if (conflicts.conflicts.length > 0) {
      await db.update('roadmap.proposals', proposal.id, {
        metadata: { ...proposal.metadata, conflicts: conflicts }
      });
      // Flag for skeptic review at D2 gate
      await escalationAdd('DEPENDENCY_UNRESOLVED', 'skeptic_d2', proposal.id, 'medium');
    }
  }
  
  return proposal;
}
```
